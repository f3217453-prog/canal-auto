name: AI Clipping

on:
  workflow_dispatch:
    inputs:
      stream_url:
        description: "URL del stream/video a clippear"
        required: true

jobs:
  clip-and-upload:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Instalar ffmpeg
        run: sudo apt-get update && sudo apt-get install -y ffmpeg

      - name: Instalar Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Instalar dependencias
        run: pip install -r requirements.txt

      - name: Instalar Ollama y descargar modelo (IA local gratis)
        run: |
          curl -fsSL https://ollama.com/install.sh | sh
          nohup ollama serve > ollama.log 2>&1 &
          for i in $(seq 1 30); do
            if curl -s http://localhost:11434 > /dev/null; then
              echo "Ollama listo"
              break
            fi
            sleep 2
          done
          ollama pull llama3.2:1b

      - name: Descargar el video fuente
        run: yt-dlp -o "input/stream.mp4" "${{ github.event.inputs.stream_url }}"

      - name: Correr el pipeline de clipping
        env:
          YOUTUBE_CLIENT_ID: ${{ secrets.YOUTUBE_CLIENT_ID }}
          YOUTUBE_CLIENT_SECRET: ${{ secrets.YOUTUBE_CLIENT_SECRET }}
          YOUTUBE_CLIPS_REFRESH_TOKEN: ${{ secrets.YOUTUBE_CLIPS_REFRESH_TOKEN }}
        run: |
          nohup ollama serve > ollama2.log 2>&1 &
          for i in $(seq 1 30); do
            if curl -s http://localhost:11434 > /dev/null; then
              echo "Ollama listo"
              break
            fi
            sleep 2
          done
          python main.py input/stream.mp4
